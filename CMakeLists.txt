cmake_minimum_required(VERSION 3.18)
project(cs259_hybrid_nn LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
find_package(OpenMP REQUIRED)

set(COMMON_SRC src/common/data_loader.cpp src/common/model.cpp src/common/timing.cpp
    src/common/validation.cpp src/serial/serial_nn.cpp src/openmp/openmp_nn.cpp)

add_executable(nn_bench src/app/main.cpp ${COMMON_SRC})
add_executable(nn_validate src/app/validate_main.cpp ${COMMON_SRC})
foreach(t nn_bench nn_validate)
  target_include_directories(${t} PRIVATE include)
  target_link_libraries(${t} PRIVATE OpenMP::OpenMP_CXX)
  target_compile_options(${t} PRIVATE -O3 -Wall -Wextra)
endforeach()

include(CheckLanguage)
check_language(CUDA)
if(CMAKE_CUDA_COMPILER)
  enable_language(CUDA)
  set(CMAKE_CUDA_STANDARD 14)
  add_executable(nn_bench_gpu src/app/main.cpp ${COMMON_SRC} src/cuda/cuda_nn.cu src/hybrid/hybrid_pipeline.cu)
  add_executable(nn_validate_gpu src/app/validate_main.cpp ${COMMON_SRC} src/cuda/cuda_nn.cu src/hybrid/hybrid_pipeline.cu)
  foreach(t nn_bench_gpu nn_validate_gpu)
    target_include_directories(${t} PRIVATE include src/cuda)
    target_compile_definitions(${t} PRIVATE USE_CUDA)
    target_link_libraries(${t} PRIVATE OpenMP::OpenMP_CXX)
    set_target_properties(${t} PROPERTIES CUDA_ARCHITECTURES native)
  endforeach()
else()
  message(STATUS "No CUDA compiler found: building CPU targets only")
endif()
